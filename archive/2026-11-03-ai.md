## 🔬 深度综合：混合检索架构 + 状态记忆 / Deep Synthesis

覆盖了 30 个 AI 概念之后，来综合两个之前分开讲过的主题：**RAG（Day 16）+ 向量数据库（Day 18）+ 推理优化（Day 46）**。

```python
# pip install openai faiss-cpu numpy

import openai
import faiss
import numpy as np
from typing import Optional

class ProductionRAGSystem:
    """
    Production-grade RAG: vector search + KV cache awareness + streaming
    Synthesizes: Embeddings + Vector DB + Inference Optimization
    """
    
    def __init__(self, dimension: int = 1536):
        # FAISS index — in production: use Pinecone/Weaviate/pgvector
        self.index = faiss.IndexFlatIP(dimension)  # Inner product (cosine sim)
        self.documents: list[str] = []
        self.client = openai.OpenAI()
    
    def embed(self, text: str) -> np.ndarray:
        response = self.client.embeddings.create(
            model="text-embedding-3-small",
            input=text
        )
        vec = np.array(response.data[0].embedding, dtype=np.float32)
        # Normalize for cosine similarity
        return vec / np.linalg.norm(vec)
    
    def add_documents(self, docs: list[str]) -> None:
        for doc in docs:
            vec = self.embed(doc)
            self.index.add(vec.reshape(1, -1))
            self.documents.append(doc)
    
    def retrieve(self, query: str, k: int = 3) -> list[str]:
        query_vec = self.embed(query)
        distances, indices = self.index.search(query_vec.reshape(1, -1), k)
        return [self.documents[i] for i in indices[0] if i < len(self.documents)]
    
    def answer(self, question: str) -> str:
        # Step 1: Retrieve relevant context
        contexts = self.retrieve(question)
        context_str = "\n\n".join(f"[{i+1}] {c}" for i, c in enumerate(contexts))
        
        # Step 2: Generate with streaming (reduces perceived latency)
        # KV cache: system prompt stays hot in cache → cheaper repeated calls
        stream = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[
                {"role": "system", "content": "Answer using only the provided context."},
                {"role": "user", "content": f"Context:\n{context_str}\n\nQuestion: {question}"}
            ],
            stream=True  # streaming for UX
        )
        
        result = ""
        for chunk in stream:
            delta = chunk.choices[0].delta.content or ""
            result += delta
            print(delta, end="", flush=True)  # stream to user
        return result

# Usage
rag = ProductionRAGSystem()
rag.add_documents([
    "Python asyncio uses epoll on Linux for I/O multiplexing.",
    "The GIL prevents true parallelism for CPU-bound tasks.",
    "ProcessPoolExecutor bypasses the GIL using separate processes.",
])
answer = rag.answer("How does Python handle concurrent I/O?")
```

**这段代码综合了什么 / What this synthesizes:**
- **Embeddings** (Day 2): 文本→向量
- **Vector DB** (Day 18): FAISS 近似最近邻搜索
- **KV Cache** (Day 46): 系统 prompt 复用热缓存
- **Streaming** (Day 46): 降低首 token 延迟感知
